## 3 · E2E 测试(本方案 P1 · 重点)

### 3.1 设计原则

1. **Page Object 模式**:每个主要页面封装成 Page Object
2. **旅程优先**:不是"页面级别",而是"用户故事级别"
3. **角色数据驱动**:用 JSON 定义 8 个角色的旅程,通过 fixtures 跑同一套测试
4. **可独立运行**:每个 spec 自包含,不依赖外部状态

### 3.3 核心 Page Object 示例

```ts
// tests/e2e/page-objects/UnifiedWorkbenchPage.ts
import { type Page, expect } from '@playwright/test'

export class UnifiedWorkbenchPage {
  constructor(private page: Page) {}

  // 角色切换
  async switchToRole(roleName: string) {
    await this.page.click('[data-testid="role-switcher-trigger"]')
    await this.page.click(`[data-testid="role-${roleName}"]`)
    await this.page.waitForURL(/\/(discovery|management|exploration)/)
  }

  // 验证快捷作业
  async expectShortcuts(expectedShortcuts: string[]) {
    const items = await this.page.locator('[data-testid="shortcut-card"]').allTextContents()
    for (const s of expectedShortcuts) {
      expect(items.some(i => i.includes(s))).toBe(true)
    }
  }

  // 验证收藏 Top 4
  async expectFavoritesCount(count: number) {
    const cards = await this.page.locator('[data-testid="favorite-card"]').count()
    expect(cards).toBeLessThanOrEqual(count)
  }

  // 验证治理全景
  async expectGovernanceVisible() {
    await expect(this.page.locator('[data-testid="governance-overview"]')).toBeVisible()
  }

  // 验证问候语
  async expectRoleGreeting(roleLabel: string) {
    await expect(this.page.locator('[data-testid="hero-greeting"]'))
      .toContainText(roleLabel)
  }
}
```

```ts
import { type Page, expect } from '@playwright/test'

export class Customer360Page {
  constructor(private page: Page) {}

  async goto(userId = 'C001') {
    await this.page.goto(`/discovery/customer360/detail/${userId}`)

    for (const f of fields) {
      expect(list.some(t => t.includes(f))).toBe(true)

    const list = await this.page.locator('[data-testid="field-permission-card"] .field-tag')
      .allTextContents()
    expect(list.some(t => t.includes(field))).toBe(false)
  }

  async favoriteCurrent() {
    await this.page.click('[data-testid="favorite-button"]')
    await expect(this.page.locator('[data-testid="favorite-button"]'))
      .toContainText('已收藏')
  }
}
```

### 8.2 E2E Spec:05-customer-360.spec.ts

```ts
import { test, expect } from '@playwright/test'
import { UnifiedWorkbenchPage } from '../page-objects/UnifiedWorkbenchPage'
import { Customer360Page } from '../page-objects/Customer360Page'

test.describe('客户 360 用户旅程', () => {
  test('张三(数据工程师):可看 L1-L3,不可看 L4', async ({ page }) => {
    const workbench = new UnifiedWorkbenchPage(page)
    const customer360 = new Customer360Page(page)

    // 1. 工作台切换角色
    await page.goto('/discovery')
    await workbench.switchToRole('数据工程师')

    // 2. 进入客户 360
    await customer360.goto('C001')

    // 3. 验证 L1-L3 字段可见
    await customer360.expectVisibleFields(['用户ID', '姓名', '身份证号', '手机号'])
    // 4. 验证 L4 字段不可见
    await customer360.expectHiddenField('征信报告')
  })

  test('收藏当前客户', async ({ page }) => {
    const customer360 = new Customer360Page(page)
    await customer360.goto('C005')
    await customer360.favoriteCurrent()

    // 验证我的收藏页能看到
    await page.goto('/management/favorites')
    await expect(page.locator('.favorite-card')).toContainText('陈军')
  })
})
```
